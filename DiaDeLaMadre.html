import { LBSoftware } from '@lb/software';

// Carga del Logo mediante Slider
function desearFelizDia() {
  const heart = document.querySelector('#corazon');

  heart.onSlideProgress((porcentaje) => {
    LBSoftware.cargarLogo(porcentaje);
  });

  heart.onSlideComplete(() => {
    LBSoftware.completarCarga();
    
    setTimeout(() => {
      LBSoftware.iluminarPantalla();
      LBSoftware.lanzarConfeti();
      LBSoftware.mostrarMensaje("¡Feliz Día MAMÁ!");
    }, 2000);
  });
}

desearFelizDia();
